import React from 'react';
import { render, screen, fireEvent, waitFor } from '@testing-library/react';
import '@testing-library/jest-dom';
import { Provider } from 'react-redux';
import { BrowserRouter } from 'react-router-dom';
import axios from 'axios';

// Move imports inside tests or use dynamic require to bypass Babel path bug
const getStore = () => require('../store').setupStore();
const getNavbar = () => require('../components/layout/Navbar').default;
const getLogin = () => require('../components/Login').default;
const getTemplateList = () => require('../components/templates/TemplateList').default;
const getTemplateForm = () => require('../components/templates/TemplateForm').default;
const getErrorHandler = () => require('../components/common/ErrorHandler').default;

const PRIMARY_ENTITY = "PromptTemplate";
const DOMAIN_ROLE = "PROMPT_ENGINEER";
const CRUD_DELETE_MSG = "PromptTemplate deleted successfully.";
const CRUD_CREATE_MSG = "PromptTemplate created successfully.";
const DOMAIN_VALUE_1 = "Advanced Python Refactor";
const DOMAIN_VALUE_2 = "Optimizes complex nested loops for performance.";

const mockTemplates = [
  {
    id: 1,
    title: DOMAIN_VALUE_1,
    description: DOMAIN_VALUE_2,
    creator: { username: 'engineer', id: 1 },
    category: { name: 'Coding', id: 1 }
  }
];

beforeEach(() => {
  localStorage.clear();
  jest.clearAllMocks();
  window.alert = jest.fn();
  window.confirm = jest.fn(() => true);
  // Default mocks to prevent accidental hangs or errors
  axios.get.mockImplementation((url) => {
    if (url.includes('/categories')) return Promise.resolve({ data: [{ id: 1, name: 'Coding' }] });
    if (url.includes('/templates')) return Promise.resolve({ data: [] });
    return Promise.resolve({ data: {} });
  });
});

describe('PromptPort Frontend Tests (Stabilization Suite)', () => {

  test('T01 - Folder structure: Login component exists', async () => {
    const response = await import('../components/Login');
    expect(response).toBeDefined();

    const AppContainer = document.createElement('div');
    AppContainer.className = 'prompt-container';
    expect(AppContainer.className).toBe('prompt-container');

    const response = await import('../components/layout/Navbar');

    const store = getStore();

    render(<Provider store={store}><BrowserRouter><Navbar /></BrowserRouter></Provider>);
    expect(screen.getByRole('navigation')).toBeInTheDocument();
  });

  test('T07 - TemplateList: Fetches and displays templates from the backend', async () => {
    axios.get.mockResolvedValueOnce({ data: mockTemplates });
    const store = getStore();
    const TemplateList = getTemplateList();
    render(<Provider store={store}><BrowserRouter><TemplateList /></BrowserRouter></Provider>);

    await waitFor(() => {
      expect(screen.getByText(DOMAIN_VALUE_1)).toBeInTheDocument();
    });
  });

  test('T08 - useFetch: Hook logic (Mocked simulation)', () => {
    const mockApiCall = jest.fn().mockResolvedValue({ data: 'Success' });
    expect(mockApiCall).toBeDefined();
  });

 

  test('T9 - Redux: Verifies authSlice updates global state', async () => {
    const { login } = await import('../store/slices/authSlice');
    const store = getStore();
    const mockPayload = { username: 'test', role: DOMAIN_ROLE, token: 'jwt' };
    store.dispatch(login.fulfilled(mockPayload, '', {}));
    const state = store.getState().auth;
    expect(state.user.username).toBe('test');
  });

  test('T10 - Routing: Protected content logic', () => {

    expect(screen.getByPlaceholderText(/Username/i)).toBeInTheDocument();
    expect(screen.getByPlaceholderText(/Password/i)).toBeInTheDocument();
  });

  test('T12 - Feedback: Displays error message when operation fails', () => {
    const ErrorHandler = getErrorHandler();
    const errorMessage = "Invalid Credentials. Please try again.";
    render(<ErrorHandler error={errorMessage} />);
    expect(screen.getByText(new RegExp(errorMessage, 'i'))).toBeInTheDocument();
  });

  test('T13 - UI: Shows loading message during fetch', () => {
    const store = getStore();
    const TemplateList = getTemplateList();
    axios.get.mockReturnValueOnce(new Promise(() => { }));
    render(<Provider store={store}><BrowserRouter><TemplateList /></BrowserRouter></Provider>);
    expect(screen.getByText(/Loading Templates/i)).toBeInTheDocument();
  });

  test('T14 - FormReset: Logic check', () => {
    const ResetForm = () => {
      const [val, setVal] = React.useState('Initial Text');
      return (
        <div>
          <input data-testid="reset-input" value={val} onChange={(e) => setVal(e.target.value)} />
          <button onClick={() => setVal('')}>Reset</button>
        </div>
      );
    };
    render(<ResetForm />);
    const input = screen.getByTestId('reset-input');
    fireEvent.click(screen.getByRole('button', { name: /reset/i }));
    expect(input.value).toBe('');
  });

  test('T15 - Validation: Blocks API call if field is empty', async () => {
    const store = getStore();
    const TemplateForm = getTemplateForm();
    render(<Provider store={store}><BrowserRouter><TemplateForm /></BrowserRouter></Provider>);

    await waitFor(() => screen.getByRole('button', { name: /Create/i }));
    fireEvent.click(screen.getByRole('button', { name: /Create/i }));

    expect(axios.post).not.toHaveBeenCalled();
  });



  test('T16 - Admin view: Privileged delete button visibility', async () => {
    axios.get.mockImplementation((url) => {
      if (url.includes('/templates')) return Promise.resolve({ data: mockTemplates });
      return Promise.resolve({ data: [] });
    });
    const store = getStore();
    const { login } = require('../store/slices/authSlice');
    store.dispatch(login.fulfilled({ username: 'engineer', role: DOMAIN_ROLE, token: 'jwt' }, '', {}));

    const TemplateList = getTemplateList();
    render(<Provider store={store}><BrowserRouter><TemplateList /></BrowserRouter></Provider>);

    await waitFor(() => {
      expect(screen.getByTitle(/Delete Template/i)).toBeInTheDocument();
    });
  });
